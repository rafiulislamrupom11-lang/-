import tkinter as tk from tkinter import filedialog, messagebox, ttk from PIL import Image, ImageTk import cv2 import os from moviepy.editor import VideoFileClip class FilmCardEditor: def __init__(self, root): self.root = root self.root.title("ফিল্ম কার্ড এডিটর - প্রোটোটাইপ") self.root.geometry("1000x700") self.root.configure(bg="#2b2b2b") self.video_path = None self.clip = None self.start_time = 0 self.end_time = None self.total_duration = 0 # UI তৈরি self.create_widgets() def create_widgets(self): # শীর্ষ বার top_frame = tk.Frame(self.root, bg="#3c3f41") top_frame.pack(fill=tk.X, padx=5, pady=5) tk.Button(top_frame, text="ওপেন ভিডিও", command=self.load_video, bg="#4caf50", fg="white").pack(side=tk.LEFT, padx=5) tk.Button(top_frame, text="এক্সপোর্ট", command=self.export_video, bg="#2196f3", fg="white").pack(side=tk.LEFT, padx=5) # প্রিভিউ এলাকা self.preview_label = tk.Label(self.root, bg="#1e1e1e") self.preview_label.pack(pady=10) # টাইমলাইন স্লাইডার control_frame = tk.Frame(self.root, bg="#2b2b2b") control_frame.pack(fill=tk.X, padx=20, pady=5) tk.Label(control_frame, text="ট্রিম স্টার্ট:", bg="#2b2b2b", fg="white").grid(row=0, column=0, sticky=tk.W) self.start_slider = tk.Scale(control_frame, from_=0, to=100, orient=tk.HORIZONTAL, length=400, bg="#2b2b2b", fg="white", command=self.update_preview) self.start_slider.grid(row=1, column=0, padx=5) tk.Label(control_frame, text="ট্রিম এন্ড:", bg="#2b2b2b", fg="white").grid(row=0, column=1, sticky=tk.W) self.end_slider = tk.Scale(control_frame, from_=0, to=100, orient=tk.HORIZONTAL, length=400, bg="#2b2b2b", fg="white", command=self.update_preview) self.end_slider.grid(row=1, column=1, padx=5) # তথ্য লেবেল self.info_label = tk.Label(self.root, text="কোনো ভিডিও লোড হয়নি", bg="#2b2b2b", fg="#cccccc") self.info_label.pack(pady=10) # প্লেব্যাক বোতাম play_btn = tk.Button(self.root, text="▶ প্রিভিউ", command=self.play_preview, bg="#ff9800", fg="white") play_btn.pack(pady=5) def load_video(self): file_path = filedialog.askopenfilename(filetypes=[("Video files", "*.mp4 *.avi *.mov *.mkv")]) if not file_path: return self.video_path = file_path self.clip = VideoFileClip(file_path) self.total_duration = self.clip.duration self.end_time = self.total_duration # স্লাইডার রেঞ্জ সেট self.start_slider.config(to=int(self.total_duration)) self.end_slider.config(to=int(self.total_duration)) self.end_slider.set(int(self.total_duration)) self.info_label.config(text=f"ভিডিও লোড: {os.path.basename(file_path)}\nডিউরেশন: {self.total_duration:.2f} সেকেন্ড") # প্রথম ফ্রেম দেখাও self.update_preview() def update_preview(self, event=None): if not self.video_path: return try: start = float(self.start_slider.get()) cap = cv2.VideoCapture(self.video_path) cap.set(cv2.CAP_PROP_POS_MSEC, start * 1000) ret, frame = cap.read() cap.release() if ret: frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB) img = Image.fromarray(frame) img.thumbnail((640, 360)) imgtk = ImageTk.PhotoImage(image=img) self.preview_label.config(image=imgtk) self.preview_label.image = imgtk except Exception as e: pass def play_preview(self): if not self.video_path: messagebox.showerror("ত্রুটি", "প্রথমে ভিডিও লোড করুন") return start = float(self.start_slider.get()) end = float(self.end_slider.get()) if start >= end: messagebox.showwarning("সতর্কতা", "শেষ সময় শুরু সময়ের চেয়ে বড় হতে হবে") return # প্রিভিউ খোলার জন্য একটি নতুন উইন্ডো (সিম্পল ওপেনসিভি প্লেব্যাক) import threading def play(): cap = cv2.VideoCapture(self.video_path) fps = cap.get(cv2.CAP_PROP_FPS) current = start cap.set(cv2.CAP_PROP_POS_MSEC, start * 1000) while current < end: ret, frame = cap.read() if not ret: break cv2.imshow("প্রিভিউ - ফিল্ম কার্ড", frame) if cv2.waitKey(int(1000/fps)) & 0xFF == ord('q'): break current += 1/fps if current >= end: break cap.release() cv2.destroyWindow("প্রিভিউ - ফিল্ম কার্ড") threading.Thread(target=play, daemon=True).start() def export_video(self): if not self.video_path: messagebox.showerror("ত্রুটি", "ভিডিও লোড করুন") return start = float(self.start_slider.get()) end = float(self.end_slider.get()) if start >= end: messagebox.showwarning("সতর্কতা", "শেষ সময় শুরু সময়ের চেয়ে বড় হতে হবে") return save_path = filedialog.asksaveasfilename(defaultextension=".mp4", filetypes=[("MP4 files", "*.mp4")]) if not save_path: return try: # MoviePy দিয়ে ট্রিম subclip = self.clip.subclipped(start, end) subclip.write_videofile(save_path, codec="libx264", audio_codec="aac") messagebox.showinfo("সফল", f"ভিডিও সেভ হয়েছে:\n{save_path}") except Exception as e: messagebox.showerror("এক্সপোর্ট ব্যর্থ", str(e)) if __name__ == "__main__": root = tk.Tk() app = FilmCardEditor(root) root.mainloop()
